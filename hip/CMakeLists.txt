# HIP port of kernel 3. Built from the top level with -DCLK_HIP=ON, or on its own:
#   cmake -S hip -B build-hip -DCMAKE_HIP_ARCHITECTURES=gfx1103            (AMD, ROCm)
#   cmake -S hip -B build-hip-nv -DCLK_HIP_NVIDIA=ON -DCMAKE_CUDA_ARCHITECTURES=89
#     NVIDIA backend, a correctness check only: the same sources compiled by nvcc against HIP's
#     header-only CUDA layer (hip-dev's include/hip/nvidia_detail), so no ROCm runtime is needed.
cmake_minimum_required(VERSION 3.21)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(clkernels_hip CXX)
  set(CMAKE_CXX_STANDARD 17)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
  endif()
  enable_testing()
endif()

option(CLK_HIP_NVIDIA "Compile the HIP sources with nvcc for an NVIDIA GPU (correctness check)" OFF)
option(CLK_HIP_RESOURCE_USAGE "Print the AMD compiler's per-kernel VGPR/SGPR/LDS/occupancy report" OFF)
if(CLK_HIP_NVIDIA)
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 17)
  set(CLK_HIP_LANG CUDA)
  find_path(CLK_HIP_INCLUDE hip/hip_runtime.h HINTS /opt/rocm/include ENV ROCM_PATH PATH_SUFFIXES include)
  if(NOT CLK_HIP_INCLUDE)
    message(FATAL_ERROR "HIP headers not found: install hip-dev or set -DCLK_HIP_INCLUDE")
  endif()
else()
  enable_language(HIP)
  set(CMAKE_HIP_STANDARD 17)
  set(CMAKE_HIP_STANDARD_REQUIRED ON)
  set(CLK_HIP_LANG HIP)
endif()
set(CLK_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/..)

# The double-precision reference and HostMatrix are shared with the OpenCL tests unchanged.
add_library(clk_hip_ref STATIC ${CLK_ROOT}/src/reference.cpp)
target_include_directories(clk_hip_ref PUBLIC ${CLK_ROOT}/include)

add_library(clk_hip STATIC gemm_hip.cpp)
set_source_files_properties(gemm_hip.cpp test_hip.cpp bench_hip.cpp PROPERTIES LANGUAGE ${CLK_HIP_LANG})
target_include_directories(clk_hip PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(clk_hip PUBLIC clk_hip_ref)
if(CLK_HIP_NVIDIA)
  target_include_directories(clk_hip PUBLIC ${CLK_HIP_INCLUDE})
  target_compile_definitions(clk_hip PUBLIC __HIP_PLATFORM_NVIDIA__)
  target_compile_options(clk_hip PUBLIC $<$<COMPILE_LANGUAGE:CUDA>:--extended-lambda>)
elseif(CLK_HIP_RESOURCE_USAGE)
  target_compile_options(clk_hip PRIVATE -Rpass-analysis=kernel-resource-usage)
endif()
